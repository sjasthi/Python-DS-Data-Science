Document formats and extraction
PDF, DOCX, and RTF emphasize document layout or formatting. The samples below run in Colab/local Python with the listed dependencies; they are not browser Run cells. Text extraction is template-specific, and a PDF is not inherently a structured data table.

DOCX — python-docx
# Colab: %pip install python-docx
from docx import Document
document = Document()
document.add_heading("Student Information",level=1)
table = document.add_table(rows=1, cols=3)
for cell,heading in zip(table.rows[0].cells,["name","email","marks"]):
    cell.text = heading
for s in students:
    for cell,value in zip(table.add_row().cells,[s.name,s.email,s.marks]):
        cell.text = str(value)
document.save("students.docx")
loaded = Document("students.docx")
rows = [[c.text for c in row.cells] for row in loaded.tables[0].rows]
docx_df = pd.DataFrame(rows[1:],columns=rows[0])
docx_df["marks"] = docx_df["marks"].astype(int)
print(docx_df)
RTF — escaped text and striprtf
# Colab: %pip install striprtf
from striprtf.striprtf import rtf_to_text
def rtf_escape(text):
    result = ""
    for ch in text:
        if ch in "\\{}": result += "\\" + ch
        elif ch == "\n": result += "\\par "
        elif ord(ch) < 128: result += ch
        else:
            for i in range(0,len(ch.encode("utf-16-le")),2):
                n = int.from_bytes(ch.encode("utf-16-le")[i:i+2],"little")
                result += "\\u" + str(n if n < 32768 else n-65536) + "?"
    return result
content = "{\\rtf1\\ansi\\uc1 " + "\\par ".join(rtf_escape(str(s)) for s in students) + "}"
with open("students.rtf","w",encoding="ascii") as f: f.write(content)
print(rtf_to_text(content))
